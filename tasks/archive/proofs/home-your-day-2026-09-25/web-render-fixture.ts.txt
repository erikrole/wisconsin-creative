function page(body: string, width: number) {
  const cssDir = path.join(ROOT, ".next/build/static/css");
  const css = readdirSync(cssDir).map((f) => readFileSync(path.join(cssDir, f), "utf8")).join("\n");
  return `<!doctype html><html lang="en"><head><meta charset="utf-8"><style>${css}
  /* Static render: settle entrance animations at their end state. */
  [style*="opacity:0"]{opacity:1!important;transform:none!important}
  body{margin:0;padding:24px;background:var(--background)} .stage{width:${width}px}</style></head>
  <body><main class="stage">${body}</main></body></html>`;
}
